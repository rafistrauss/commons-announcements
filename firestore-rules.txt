rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() {
      return request.auth != null;
    }

    function normalizedEmail() {
      return lower(request.auth.token.email);
    }

    function isDefaultAdmin() {
      return normalizedEmail() == "your-email@example.com";
    }

    function isListedAdmin() {
      return exists(/databases/$(database)/documents/admin-users/$(normalizedEmail()));
    }

    function isAdmin() {
      return signedIn() && (isDefaultAdmin() || isListedAdmin());
    }

    match /aliyot-signups/{docId} {
      allow create: if true;
      allow read, update, delete: if isAdmin();
    }

    match /admin-users/{emailDocId} {
      allow get: if signedIn() && emailDocId == normalizedEmail();
      allow list, create, update, delete: if isAdmin();
    }
  }
}
